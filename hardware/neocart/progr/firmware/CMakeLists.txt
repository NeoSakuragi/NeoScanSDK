cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD neocart_progr CACHE STRING "board")
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "platform")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(neocart_progr C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(neocart_progr neocart_progr.c)
pico_enable_stdio_usb(neocart_progr 1)
pico_enable_stdio_uart(neocart_progr 0)
target_link_libraries(neocart_progr pico_stdlib hardware_gpio hardware_adc)
pico_add_extra_outputs(neocart_progr)
